java.nio.file.Files — the modern way

// Reading Path path = Path.of("data.txt"); String all = Files.readString(path); // entire file as String byte[] bytes = Files.readAllBytes(path); // entire file as bytes List<String> lines = Files.readAllLines(path); // all lines into List // Streaming — memory-efficient for large files try (Stream<String> stream = Files.lines(path)) { stream.filter(l -> !l.isBlank()) .map(String::trim) .forEach(System.out::println); } // Specify charset explicitly Files.readString(path, StandardCharsets.UTF_8);
// Writing Path out = Path.of("output.txt"); Files.writeString(out, "Hello, World!"); Files.write(out, List.of("line 1", "line 2")); // Append mode Files.writeString(out, "new line\n", StandardOpenOption.APPEND); // Copy / Move Files.copy(src, dst, StandardCopyOption.REPLACE_EXISTING); Files.move(src, dst, StandardCopyOption.ATOMIC_MOVE); // Delete Files.delete(path); // throws if not found Files.deleteIfExists(path); // safe

Directory Operations

// Create directories Files.createDirectory(Path.of("newdir")); // parent must exist Files.createDirectories(Path.of("a/b/c")); // creates all parents // List directory contents try (Stream<Path> stream = Files.list(Path.of("."))) { stream.filter(Files::isRegularFile) .forEach(System.out::println); } // Walk directory tree recursively try (Stream<Path> walk = Files.walk(Path.of("src"))) { walk.filter(p -> p.toString().endsWith(".java")) .forEach(System.out::println); }
// File metadata Path p = Path.of("file.txt"); Files.exists(p) Files.isReadable(p) Files.isWritable(p) Files.isDirectory(p) Files.isRegularFile(p) Files.size(p) // bytes Files.getLastModifiedTime(p) // Temp files Path tmp = Files.createTempFile("prefix", ".txt"); Path tmpDir = Files.createTempDirectory("myapp"); // Remember to delete temp files on exit tmp.toFile().deleteOnExit();

Readers & Writers — character-based (text files)

// BufferedReader — read text line by line Path path = Path.of("data.txt"); try (BufferedReader br = Files.newBufferedReader(path, StandardCharsets.UTF_8)) { String line; while ((line = br.readLine()) != null) { System.out.println(line); } } // As Stream (preferred modern way) try (BufferedReader br = Files.newBufferedReader(path)) { br.lines() .filter(l -> l.startsWith("#")) .forEach(System.out::println); }
// BufferedWriter — write text try (BufferedWriter bw = Files.newBufferedWriter( Path.of("output.txt"), StandardCharsets.UTF_8, StandardOpenOption.CREATE)) { bw.write("Line 1"); bw.newLine(); // platform-correct newline bw.write("Line 2"); } // PrintWriter — formatted output try (PrintWriter pw = new PrintWriter( Files.newBufferedWriter(Path.of("out.txt")))) { pw.printf("%-10s %5d%n", "Alice", 42); pw.println("done"); }

Reading classpath resources (inside a JAR)

// Files.of() doesn't work inside JARs — use getResourceAsStream() try (InputStream is = getClass().getResourceAsStream("/config.properties"); BufferedReader br = new BufferedReader(new InputStreamReader(is, StandardCharsets.UTF_8))) { Properties props = new Properties(); props.load(br); String host = props.getProperty("db.host"); } // Spring: @Value from application.properties — even simpler @Value("${db.host}") private String host;

java.nio.file.Path — cross-platform file paths

// Creating Paths Path p1 = Path.of("src/main/java"); // Java 11+ Path p2 = Paths.get("src", "main", "java"); // Java 7 Path abs = Path.of("/home/user/docs"); Path home = Path.of(System.getProperty("user.home")); // Path operations — all return new Path p1.resolve("MyClass.java") // append p1.getParent() // src/main p1.getFileName() // java (as Path) p1.getRoot() // null (relative path) p1.toAbsolutePath() // absolute path p1.normalize() // resolve .. and . p1.relativize(p2) // relative from p2 to p1
// Path comparison and info Path a = Path.of("data/file.txt"); a.toString() // "data/file.txt" (or \\ on Windows) a.getNameCount() // 2 a.getName(0) // data a.getName(1) // file.txt a.startsWith("data") a.endsWith("txt") // Glob matching PathMatcher pm = FileSystems.getDefault() .getPathMatcher("glob:**.java"); pm.matches(a); // false — not .java // Convert to File (legacy interop) File f = a.toFile(); Path back = f.toPath();

Java Serialization — converting objects to bytes

// Implement Serializable to enable class User implements Serializable { private static final long serialVersionUID = 1L; private String name; private int age; transient private String password; // NOT serialized } // Serialize (write) try (ObjectOutputStream oos = new ObjectOutputStream( new FileOutputStream("user.ser"))) { oos.writeObject(new User("Alice", 30)); }
// Deserialize (read) try (ObjectInputStream ois = new ObjectInputStream( new FileInputStream("user.ser"))) { User user = (User) ois.readObject(); } // Modern alternatives — prefer these: // - Jackson/Gson → JSON // - Protocol Buffers → binary, typed // - Kryo → fast binary serialization // - Java Records → immutable, serializable
Java serialization has known security vulnerabilities (deserialization attacks). For new code, prefer JSON (Jackson) or binary formats (Protobuf). Never deserialize data from untrusted sources without careful filtering.
