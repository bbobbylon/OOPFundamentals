Insert (sift up) & Extract-Min (sift down)

// Insert: add to end, sift UP void insert(int val) { heap.add(val); siftUp(heap.size() - 1); } void siftUp(int i) { while (i > 0) { int parent = (i - 1) / 2; if (heap[i] < heap[parent]) { swap(i, parent); // bubble up i = parent; } else break; // heap property restored } }
// Extract-min: swap root with last, remove, sift DOWN int extractMin() { int min = heap[0]; heap[0] = heap.remove(heap.size() - 1); siftDown(0); return min; } void siftDown(int i) { int n = heap.size(); while (2*i+1 < n) { int j = 2*i+1; // left child if (j+1<n && heap[j+1]<heap[j]) j++; // pick smaller child if (heap[i] <= heap[j]) break; swap(i, j); i = j; } }

Heap Sort — O(n log n) time, O(1) space

// Phase 1: Build max-heap from array — O(n) // Start from last non-leaf and sift down void buildMaxHeap(int[] arr) { for (int i = arr.length/2 - 1; i >= 0; i--) { siftDown(arr, i, arr.length); } } // Phase 2: Extract elements in order — O(n log n) void heapSort(int[] arr) { buildMaxHeap(arr); // max at arr[0] for (int end = arr.length - 1; end > 0; end--) { swap(arr, 0, end); // move max to end siftDown(arr, 0, end); // restore heap on reduced array } // arr is now sorted ascending! } // Why not used in practice? // - Poor cache performance (sifting jumps around memory) // - Not stable // - Quicksort is faster in practice // - Merge sort is stable and nearly as fast // - Java Arrays.sort() uses TimSort (merge + insertion)

Java PriorityQueue — min-heap by default

// Min-heap (natural ordering) PriorityQueue<Integer> minHeap = new PriorityQueue<>(); minHeap.offer(5); // add minHeap.offer(2); minHeap.offer(8); minHeap.peek(); // 2 — view min, no remove minHeap.poll(); // 2 — remove and return min minHeap.size(); // 2 // Max-heap — reverse comparator PriorityQueue<Integer> maxHeap = new PriorityQueue<>(Collections.reverseOrder()); // Or with Comparator.reverseOrder() PriorityQueue<Integer> max2 = new PriorityQueue<>((a, b) -> b - a);
// Custom objects — sort by field PriorityQueue<Task> tasks = new PriorityQueue<>( Comparator.comparingInt(Task::getPriority) ); tasks.offer(new Task("low", 5)); tasks.offer(new Task("urgent", 1)); tasks.offer(new Task("medium", 3)); tasks.poll(); // Task{urgent, priority=1} // Note: iteration order is NOT guaranteed sorted // Only poll() gives sorted order // PQ doesn't allow O(1) search or update-key
