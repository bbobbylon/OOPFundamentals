Integer vs Floating-Point Division

// Integer division — truncates toward zero int a = 10 / 3; // 3 (not 3.333) int b = -7 / 2; // -3 (truncated, not -4) int c = 10 / 0; // ArithmeticException! // Modulo with negatives int d = -7 % 3; // -1 (sign follows dividend) int e = 7 % -3; // 1
// Floating-point division — never throws double x = 10.0 / 3; // 3.3333... double y = 10 / 3; // 3.0! (int/int first) double z = 10.0 / 0; // Infinity (not exception!) double w = 0.0 / 0; // NaN // Force double division int a = 10, b = 3; double r = (double) a / b; // 3.333...

Pre vs Post Increment — subtle difference in expressions

int i = 5; int a = i++; // a = 5, then i becomes 6 (post: use then increment) int b = ++i; // i becomes 7, then b = 7 (pre: increment then use) // In standalone statements, both are identical: i++; // same as ++i; // same as i = i + 1 // String + int — concatenation wins left to right System.out.println("val: " + 1 + 2); // "val: 12" (not 3!) System.out.println("val: " + (1 + 2)); // "val: 3"

Short-Circuit Evaluation — crucial for null-safety

// && stops at first false String s = null; if (s != null && s.length() > 0) { // safe! s.length() never called if s==null } // || stops at first true boolean result = isAdmin(user) || hasPermission(user); // hasPermission() never called if isAdmin is true
// Non-short-circuit: & and | always evaluate both boolean x = false & risky(); // risky() IS called boolean y = true | risky(); // risky() IS called // Use case: when side effects are needed if (writeA() | writeB()) { // both always write ... } // In practice: nearly always use && and ||

Practical Bitwise Uses

// Bit flags — store multiple boolean flags in one int static final int READ = 0b001; // 1 static final int WRITE = 0b010; // 2 static final int EXECUTE = 0b100; // 4 int perms = READ | WRITE; // 0b011 = 3 // Check a flag if ((perms & READ) != 0) { // has READ? // Add a flag perms = perms | EXECUTE; // Remove a flag perms = perms & ~WRITE;
// Fast multiply/divide by powers of 2 int x = 5; int mul8 = x << 3; // 40 (5 * 8) int div4 = x >> 2; // 1 (5 / 4, truncated) // Check odd/even without % boolean isOdd = (x & 1) == 1; // true if last bit is 1 // Swap without temp variable int a = 5, b = 3; a = a ^ b; b = a ^ b; a = a ^ b; // a=3, b=5

Ternary Operator — compact if-else expression

// Syntax: condition ? valueIfTrue : valueIfFalse int x = 10; String label = (x > 5) ? "big" : "small"; // "big" // Can be nested (but hard to read — prefer if-else for complex logic) String size = x > 100 ? "huge" : x > 50 ? "large" : x > 10 ? "medium" : "small"; // Use in method calls System.out.println(user != null ? user.getName() : "Guest"); // Common: Math.max/min equivalent int max = (a > b) ? a : b; int abs = (n < 0) ? -n : n;

instanceof — type checking + pattern matching (Java 16+)

// Classic instanceof (Java 1.0) Object obj = "hello"; if (obj instanceof String) { String s = (String) obj; // cast needed System.out.println(s.length()); } // Pattern matching instanceof (Java 16+) if (obj instanceof String s) { System.out.println(s.length()); // s auto-cast! } // Use in conditions if (obj instanceof String s && s.length() > 3) { System.out.println(s.toUpperCase()); }

Common Precedence Pitfalls

// Pitfall 1: == vs && boolean x = true, y = false; if (x == true && y == false) { // == before && // Pitfall 2: + concatenation priority System.out.println(1 + 2 + " hi"); // "3 hi" (left to right, 1+2=3 first) System.out.println("hi " + 1 + 2); // "hi 12" (string concat left to right) System.out.println("hi " + (1+2)); // "hi 3" (force addition first) // Pitfall 3: assignment in condition int n; if ((n = getValue()) != 0) { // assign then compare System.out.println(n); }
