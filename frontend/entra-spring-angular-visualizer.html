// app.config.ts — MSAL Angular setup (Angular 17+) import { MsalModule, MsalInterceptor, MsalGuard } from '@azure/msal-angular'; import { PublicClientApplication, InteractionType } from '@azure/msal-browser'; const msalInstance = new PublicClientApplication({ auth: { clientId: '', authority: 'https://login.microsoftonline.com/', redirectUri: 'http://localhost:4200', }, cache: { cacheLocation: 'sessionStorage' } }); export const appConfig: ApplicationConfig = { providers: [ provideRouter(routes), provideHttpClient(withInterceptorsFromDi()), importProvidersFrom( MsalModule.forRoot(msalInstance, { interactionType: InteractionType.Redirect, authRequest: { scopes: ['api:///Users.Read'] } }, { interactionType: InteractionType.Redirect, protectedResourceMap: new Map([ ['http://localhost:8080/api/', ['api:///Users.Read']] ]) }) ), { provide: HTTP_INTERCEPTORS, useClass: MsalInterceptor, multi: true } ] };
// app.routes.ts — protect routes with MsalGuard import { MsalGuard } from '@azure/msal-angular'; export const routes: Routes = [ { path: '', component: HomeComponent }, { path: 'dashboard', component: DashboardComponent, canActivate: [MsalGuard] // triggers login if no session }, { path: 'admin', component: AdminComponent, canActivate: [MsalGuard, RoleGuard], // custom role check data: { requiredRole: 'Admin' } } ]; // role.guard.ts — check Entra app roles from ID token @Injectable({ providedIn: 'root' }) export class RoleGuard implements CanActivate { constructor(private msal: MsalService) {} canActivate(route: ActivatedRouteSnapshot): boolean { const required = route.data['requiredRole']; const account = this.msal.instance.getActiveAccount(); const roles = account?.idTokenClaims?.['roles'] as string[] ?? []; return roles.includes(required); } }
// The MsalInterceptor handles token attachment automatically. // For manual control, use MsalService: @Injectable({ providedIn: 'root' }) export class UserService { constructor( private http: HttpClient, private msal: MsalService ) {} getUsers(): Observable<User[]> { // MsalInterceptor sees the URL matches protectedResourceMap // and automatically adds: Authorization: Bearer return this.http.get<User[]>('http://localhost:8080/api/users'); } getCurrentUser(): AccountInfo | null { return this.msal.instance.getActiveAccount(); } logout(): void { this.msal.logoutRedirect(); } }
// SecurityConfig.java — Spring Boot Resource Server @Configuration @EnableMethodSecurity public class SecurityConfig { @Bean public SecurityFilterChain filterChain(HttpSecurity http) throws Exception { http .cors(cors -> cors.configurationSource(corsConfigurationSource())) .csrf(AbstractHttpConfigurer::disable) // stateless API .authorizeHttpRequests(auth -> auth .requestMatchers("/actuator/health").permitAll() .anyRequest().authenticated() ) .oauth2ResourceServer(oauth2 -> oauth2 .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtConverter())) ); return http.build(); } @Bean public JwtAuthenticationConverter jwtConverter() { JwtGrantedAuthoritiesConverter conv = new JwtGrantedAuthoritiesConverter(); conv.setAuthoritiesClaimName("roles"); // Entra app roles claim conv.setAuthorityPrefix("APPROLE_"); // → APPROLE_Admin JwtAuthenticationConverter jwtConv = new JwtAuthenticationConverter(); jwtConv.setJwtGrantedAuthoritiesConverter(conv); return jwtConv; } @Bean public CorsConfigurationSource corsConfigurationSource() { CorsConfiguration cfg = new CorsConfiguration(); cfg.setAllowedOrigins(List.of("http://localhost:4200")); cfg.setAllowedMethods(List.of("GET","POST","PUT","DELETE")); cfg.setAllowedHeaders(List.of("*")); var source = new UrlBasedCorsConfigurationSource(); source.registerCorsConfiguration("/**", cfg); return source; } }
// UserController.java — using JWT claims for authz @RestController @RequestMapping("/api/users") public class UserController { @GetMapping @PreAuthorize("hasAuthority('SCOPE_Users.Read') or hasAuthority('APPROLE_Admin')") public List<User> getUsers() { ... } @DeleteMapping("/{id}") @PreAuthorize("hasAuthority('APPROLE_Admin')") public void deleteUser(@PathVariable Long id) { ... } // Access raw JWT claims in controller @GetMapping("/me") public Map<String, Object> whoAmI(@AuthenticationPrincipal Jwt jwt) { return Map.of( "email", jwt.getClaimAsString("preferred_username"), "name", jwt.getClaimAsString("name"), "roles", jwt.getClaimAsStringList("roles") ); } }

3 — Integration Checklist

Check off each step as you set up the integration. Click any item for details:

1
Register the Angular SPA in Entra ID
Create App Registration → Authentication → Add SPA platform → add redirect URI (localhost:4200, prod URL). Select "Access tokens" only if you use implicit flow (not needed with PKCE).
2
Register the Spring Boot API in Entra ID
Create a second App Registration for the API. In "Expose an API", define scopes like Users.Read. Note the Application ID URI (e.g., api://<client-id>).
3
Add API permissions to the Angular registration
In the Angular App Registration → API permissions → Add a permission → My APIs → select the Spring Boot API registration → select the scopes. Grant admin consent.
4
Configure Angular with MSAL library
Install @azure/msal-angular @azure/msal-browser. Configure MsalModule with Angular SPA's client ID, tenant, and the API scope in protectedResourceMap.
5
Configure Spring Boot as Resource Server
Add spring-boot-starter-oauth2-resource-server dependency. Set spring.security.oauth2.resourceserver.jwt.issuer-uri to the Entra v2 endpoint. Set audiences to the API's client ID.
6
Configure CORS on Spring Boot
Spring must allow the Angular origin (localhost:4200 in dev, prod URL in production). Configure via CorsConfigurationSource bean or @CrossOrigin annotations. Without this, browsers block all API responses.
7
Test the full flow end to end
Open Angular app → click login → Entra login page → consent if first time → redirect back → check MSAL stores a token → open browser DevTools Network → find an API call → confirm Authorization: Bearer header present → confirm API returns 200.
