TypeScript Async Patterns

Promises, async/await, all combinators, async iterables, error handling, and real-world patterns — with TypeScript types throughout

What is async JavaScript? Why do we need it?

JavaScript runs on a single thread. If you make a network request (fetch an API, read a file) and wait for it to finish before moving on, the entire page freezes — no clicks, no scrolls, nothing. Asynchronous programming lets JavaScript start an operation, register a callback to run when it finishes, and continue doing other work in the meantime.

A Promise is JavaScript's object for "a value that will exist in the future." It starts as pending, then either resolves (success, you get the value) or rejects (failure, you get an error). async/await is syntactic sugar that makes Promise code look synchronous — the await keyword pauses only that async function, never the whole thread.

Promise vs async/await
// Promises (callback-style):
fetch(url).then(r => r.json()).then(data => log(data));

// async/await (reads sequentially):
const r = await fetch(url);
const data = await r.json();
TypeScript adds types to async
  • Promise<User> — typed result
  • async fn returns Promise<T>
  • Awaited<Promise<User>> = User
  • catch (e: unknown) — safe error handling
In Angular / Spring / CIAM work: Angular HTTP calls return Observable (RxJS), but many APIs use Promises. CIAM auth flows are async: await oidc.getToken(), await http.get<User>(url). Spring Boot's WebFlux uses Reactor (similar concept). Promise.all() for parallel auth checks.

// TypeScript: Promise<T> — T is the resolved type
const p: Promise<number> = new Promise((resolve, reject) => {
  setTimeout(() => {
    if (Math.random() > 0.5) resolve(42);   // fulfilled: T=42
    else                    reject(new Error('failed')); // rejected
  }, 1000);
});

// .then(onFulfilled, onRejected) — chainable
p.then(n => n * 2)       // Promise<number>
  .then(n => n.toString()) // Promise<string>
  .catch(e => 'default')  // catches any earlier rejection
  .finally(() => cleanup()); // always runs

// Shorthand creators
Promise.resolve(42);      // already fulfilled: Promise<number>
Promise.reject(new Error('bad')); // already rejected

async / await — syntactic sugar over Promises

// The two are EXACTLY equivalent

// Promise chain style
function getUser(id: number): Promise<User> {
  return fetch(`/users/${id}`)
    .then(r => r.json())
    .then(data => data as User)
    .catch(e => { throw new Error('fetch failed'); });
}

// async/await style — reads like sync code
async function getUser(id: number): Promise<User> {
  try {
    const r = await fetch(`/users/${id}`);
    return await r.json() as User;
  } catch {
    throw new Error('fetch failed');
  }
}

// async function ALWAYS returns Promise<T>
async function four(): Promise<number> { return 4; }
// ↑ Same as: function four() { return Promise.resolve(4); }

// await unwraps the Promise — only inside async fn
const n = await four(); // n: number (4)

// Classic use: timeout wrapper

function withTimeout<T>(
  p: Promise<T>,
  ms: number
): Promise<T> {
  const timeout = new Promise<never>((_, reject) =>
    setTimeout(() => reject(new Error(`Timeout after ${ms}ms`)), ms)
  );
  return Promise.race([p, timeout]);
}

await withTimeout(fetchData(), 5000);
// → resolves with data if < 5s, rejects with timeout if not

// ✓ Best for: timeout, fastest-server wins, cache vs network

Async Error Handling Patterns

The Result type — no-throw pattern

// Never throw async — return a discriminated union
type Result<T, E = Error> =
  | { ok: true;  value: T }
  | { ok: false; error: E };

async function safeGet<T>(
  url: string
): Promise<Result<T>> {
  try {
    const r = await fetch(url);
    if (!r.ok) throw new Error(`HTTP ${r.status}`);
    return { ok: true, value: await r.json() };
  } catch (e) {
    return { ok: false, error: e as Error };
  }
}

// Usage — explicit branching, no surprise throws
const result = await safeGet<User>('/api/me');
if (result.ok) {
  console.log(result.value.name); // TS: User
} else {
  console.error(result.error.message);
}

Retry with exponential backoff

async function retry<T>(
  fn: () => Promise<T>,
  opts = { times: 3, baseMs: 500 }
): Promise<T> {
  let lastError: unknown;
  for (let i = 0; i < opts.times; i++) {
    try {
      return await fn();
    } catch (e) {
      lastError = e;
      if (i < opts.times - 1) {
        const delay = opts.baseMs * 2 ** i; // 500, 1000, 2000...
        await new Promise(r => setTimeout(r, delay));
      }
    }
  }
  throw lastError;
}

// Usage
const data = await retry(
  () => fetch('/flaky-api').then(r => r.json()),
  { times: 4, baseMs: 300 }
);

Async Iterables — for await...of

Creating & consuming async iterables

// AsyncGenerator — yields one value at a time, lazily
async function* paginate<T>(
  url: string
): AsyncGenerator<T[]> {
  let page = 1;
  while (true) {
    const res = await fetch(`${url}?page=${page}`);
    const data: { items: T[]; hasMore: boolean } =
      await res.json();
    yield data.items;                // pause, emit this page
    if (!data.hasMore) break;       // done!
    page++;
  }
}

// Consume with for await...of
for await (const page of paginate<User>('/api/users')) {
  page.forEach(user => renderUser(user));
  // Each page loads before the next fetch fires
}

// With break — stops fetching early
let found: User | undefined;
for await (const page of paginate<User>('/api/users')) {
  found = page.find(u => u.name === 'Bobby');
  if (found) break; // no more pages fetched!
}

Async iterable from EventTarget / stream

// Turn any event into an async iterable
async function* fromEvent<T>(
  target: EventTarget,
  event: string
): AsyncGenerator<T> {
  const queue: T[] = [];
  let resolve: (() => void) | null = null;

  target.addEventListener(event, (e) => {
    queue.push(e as T);
    resolve?.(); resolve = null;
  });

  while (true) {
    if (queue.length) { yield queue.shift()!; }
    else { await new Promise<void>(r => { resolve = r; }); }
  }
}

// Consume clicks as an async stream
for await (const click of fromEvent<MouseEvent>(btn, 'click')) {
  console.log('Clicked at', click.clientX, click.clientY);
}

// ReadableStream (Fetch API / Node streams)
const res = await fetch('/api/large-file');
const reader = res.body!.getReader();
const decoder = new TextDecoder();
while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  processChunk(decoder.decode(value));
}

TypeScript async types — Awaited<T>, return type inference, typed errors

Awaited<T> — unwrap the resolved type

// Awaited extracts the value type from a Promise
type A = Awaited<Promise<string>>;     // string
type B = Awaited<Promise<Promise<number>>>; // number (nested)
type C = Awaited<string>;               // string (no-op on non-Promise)

// Common use: extract the return type of an async fn
async function fetchUser(id: string) {
  const res = await fetch(`/api/users/${id}`);
  return res.json() as User;
}

// ReturnType gets the Promise; Awaited unwraps it
type UserResult = Awaited<ReturnType<typeof fetchUser>>;
// UserResult = User  ← not Promise<User>

// Useful in generics that accept async or sync callbacks
async function run<T>(fn: () => T | Promise<T>): Promise<T> {
  return await fn();  // Awaited<T> normalizes T in the caller
}

Typed errors in catch — unknown vs cast

// TypeScript: catch variable is always  unknown
try {
  await doSomething();
} catch (e) {
  // e: unknown — must narrow before use
  if (e instanceof Error) {
    console.log(e.message); // ✓ narrow first
  }
}

// Custom error type for typed error handling
class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
    this.name = 'ApiError';
  }
}

// Error guard helper — use everywhere
function isApiError(e: unknown): e is ApiError {
  return e instanceof ApiError;
}

try { await api.getUser(); }
catch (e) {
  if (isApiError(e) && e.status === 401) {
    redirectToLogin();
  } else throw e; // re-throw unexpected errors
}

Promise.any() — first success wins

// any() resolves on first fulfilled promise
// only rejects if ALL reject (AggregateError)
// contrast: race() resolves/rejects on first settled
const fastest = await Promise.any([
  fetchFromCdn1(url),  // ← takes 200ms ✓ wins
  fetchFromCdn2(url),  // takes 400ms
  fetchFromOrigin(url) // takes 150ms but rejects
]);

// Type: typeof fastest  →  same as the element type
// AggregateError if all reject:
try { await Promise.any([Promise.reject(1), Promise.reject(2)]); }
catch (e) {
  if (e instanceof AggregateError) {
    console.log(e.errors); // [1, 2]
  }
}

Async generics — typed async pipelines

// Fully typed async transform pipeline
async function pipe<A, B, C>(
  value: A,
  step1: (a: A) => Promise<B>,
  step2: (b: B) => Promise<C>
): Promise<C> {
  return step2(await step1(value));
}

// Use: all types inferred
const result = await pipe(
  'user-123',
  id => fetchUser(id),   // string → Promise<User>
  u  => fetchProfile(u)  // User → Promise<Profile>
);
// result: Profile — fully inferred, no casts

Production Patterns

Concurrent limit (p-limit style)

// Run max N promises concurrently
async function mapWithLimit<T, R>(
  items: T[],
  fn: (item: T) => Promise<R>,
  limit: number
): Promise<R[]> {
  const results: R[] = [];
  let i = 0;

  async function worker() {
    while (i < items.length) {
      const idx = i++;
      results[idx] = await fn(items[idx]);
    }
  }

  await Promise.all(
    Array.from({ length: limit }, worker)
  );
  return results;
}

// Upload 100 files, max 5 at a time
const urls = await mapWithLimit(
  files, f => uploadFile(f), 5
);

AbortController — cancel in-flight requests

// Cancel a fetch when component unmounts (Angular)
@Component({...})
export class SearchComponent implements OnDestroy {
  private controller: AbortController | null = null;

  async search(q: string) {
    // Cancel previous request
    this.controller?.abort();
    this.controller = new AbortController();

    try {
      const res = await fetch(`/search?q=${q}`, {
        signal: this.controller.signal  // pass the signal
      });
      this.results = await res.json();
    } catch (e) {
      if ((e as DOMException).name === 'AbortError') return;
      throw e; // real error — rethrow
    }
  }

  ngOnDestroy() { this.controller?.abort(); }
}
