Custom Form Controls — ControlValueAccessor

Build any custom UI component that integrates seamlessly with Angular's reactive forms and template-driven forms — no wrapper needed.

1. Why ControlValueAccessor Exists

Angular's reactive forms know how to talk to native HTML inputs (<input>, <select>, <textarea>) through built-in value accessors. When you build a custom component — a star rating, a color picker, a chip list — the form system has no idea how to read or write its value. ControlValueAccessor is the bridge.

Without CVA

// You'd have to manually sync everything
<star-rating
  [value]="form.get('rating')?.value"
  (valueChange)="form.get('rating')?.setValue($event)"
  (blur)="form.get('rating')?.markAsTouched()"
></star-rating>

// Error messages only work if you manually
// check the FormControl too
// Disabled state? Another manual binding...
// This gets messy fast with complex forms.

With CVA

// Works exactly like a native input!

// Template-driven
<star-rating [(ngModel)]="product.rating"></star-rating>

// Reactive forms
<star-rating formControlName="rating"></star-rating>

// Disabled state propagated automatically
this.form = new FormGroup({
  rating: new FormControl({ value: 3, disabled: false })
});
// Validators work too: Validators.required,
// Validators.min(1), custom validators

2. The CVA Interface — 4 Methods

MethodCalled by Angular when…Your job
writeValue(val) Form sets the value (e.g. setValue(), patch, initial value) Update your component's visual state to show val
registerOnChange(fn) Angular sets up the form control Save fn and call it whenever user changes value
registerOnTouched(fn) Angular sets up the form control Save fn and call it when user leaves (blur)
setDisabledState(disabled) Form control is enabled/disabled Apply disabled visual + behavior to your component 
import { ControlValueAccessor, NG_VALUE_ACCESSOR } from '@angular/forms';
import { Component, forwardRef } from '@angular/core';

@Component({
  selector: 'app-custom-input',
  standalone: true,
  providers: [{
    provide: NG_VALUE_ACCESSOR,       // ← register as CVA
    useExisting: forwardRef(() => CustomInputComponent),
    multi: true                        // ← always multi:true
  }]
})
export class CustomInputComponent
  implements ControlValueAccessor {

  value: any = null;
  isDisabled = false;

  // Callbacks Angular registers — call these to notify the form
  private onChange: (v: any) => void = () => {};
  private onTouched: () => void        = () => {};

  // Angular → component
  writeValue(val: any) { this.value = val; }

  // Component → Angular
  registerOnChange(fn: (v: any) => void) { this.onChange = fn; }
  registerOnTouched(fn: () => void)        { this.onTouched = fn; }
  setDisabledState(d: boolean)             { this.isDisabled = d; }

  // Called from template when user interacts
  handleChange(v: any) {
    this.value = v;
    this.onChange(v);   // MUST call — notifies form
  }
  handleBlur() { this.onTouched(); }
}

4. Example 1 — Star Rating Component

Interactive Demo

Click a star to set the rating. Hover to preview. The form control below reflects the value.

★ ★ ★ ★ ★
No rating selected
formControl.value: null
formControl.status: INVALID
formControl.dirty: false
required: fail min(1): fail 
// star-rating.component.ts
@Component({
  selector: 'app-star-rating',
  standalone: true,
  providers: [{
    provide: NG_VALUE_ACCESSOR,
    useExisting: forwardRef(() => StarRatingComponent),
    multi: true
  }],
  template: `
    <div class="stars" [class.disabled]="isDisabled"
         (blur)="onTouched()">
      <span *ngFor="let n of [1,2,3,4,5]"
            [class.lit]="n <= value"
            (click)="!isDisabled && select(n)"
            (mouseenter)="hover = n"
            (mouseleave)="hover = 0">
        ★
      </span>
    </div>`
})
export class StarRatingComponent
  implements ControlValueAccessor {

  value: number | null = null;
  hover = 0;
  isDisabled = false;

  private onChange = (_: number | null) => {};
  private onTouched = () => {};

  writeValue(v: number | null) { this.value = v; }
  registerOnChange(fn: any)    { this.onChange = fn; }
  registerOnTouched(fn: any)   { this.onTouched = fn; }
  setDisabledState(d: boolean) { this.isDisabled = d; }

  select(n: number) {
    this.value = n;
    this.onChange(n);   // ← notify the form
    this.onTouched();    // ← mark as touched
  }
}

// Usage in a reactive form
form = new FormGroup({
  rating: new FormControl(null, [
    Validators.required,
    Validators.min(1)
  ])
});
// Template: <app-star-rating formControlName="rating"></app-star-rating>

5. Example 2 — Phone Number Input with Formatting

Live Demo

Type a US phone number. The component automatically formats it as (555) 123-4567 and returns only digits to the form.

Please enter a complete 10-digit phone number

formatted display: —

// phone-input.component.ts
@Component({
  selector: 'app-phone-input',
  standalone: true,
  providers: [
    { provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(() => PhoneInputComponent), multi: true },
    { provide: NG_VALIDATORS, useExisting: forwardRef(() => PhoneInputComponent), multi: true }
  ],
  template: `
    <div class="phone-wrapper">
      <span class="flag">🇺🇸</span>
      <input #input type="tel"
        [value]="displayValue"
        (input)="onInput($event)"
        (blur)="onTouched()" />
    </div>`
})
export class PhoneInputComponent
  implements ControlValueAccessor, Validator {

  displayValue = '';       // (555) 123-4567
  rawValue     = '';       // 5551234567 — what the form gets

  writeValue(v: string | null) {
    this.rawValue = v ?? '';
    this.displayValue = this.format(this.rawValue);
  }

  onInput(e: Event) {
    const digits = (e.target as HTMLInputElement).value
      .replace(/\D/g, '').slice(0, 10);
    this.rawValue = digits;
    this.displayValue = this.format(digits);
    this.onChange(digits);    // form gets raw digits
  }

  format(digits: string): string {
    if (digits.length < 4)  return digits;
    if (digits.length < 7)  return `(${digits.slice(0,3)}) ${digits.slice(3)}`;
    return `(${digits.slice(0,3)}) ${digits.slice(3,6)}-${digits.slice(6)}`;
  }

  // Validator interface
  validate(ctrl: AbstractControl): ValidationErrors | null {
    return ctrl.value?.length === 10
      ? null
      : { phoneInvalid: true };
  }

  private onChange   = (_: string) => {};
  private onTouched  = () => {};
  registerOnChange(fn: any)  { this.onChange = fn; }
  registerOnTouched(fn: any) { this.onTouched = fn; }
  setDisabledState(d: boolean) { this.isDisabled = d; }
}

6. Example 3 — Chip List Multi-Select

Live Demo

Type a technology and press Enter or comma, or click a suggestion. The form control receives a string[].

formControl.value: []
// chip-select.component.ts
@Component({
  selector: 'app-chip-select',
  standalone: true,
  providers: [{
    provide: NG_VALUE_ACCESSOR,
    useExisting: forwardRef(() => ChipSelectComponent),
    multi: true
  }],
  template: `
    <div class="chip-area">
      <span *ngFor="let tag of selected" class="chip">
        {{ tag }}
        <button (click)="remove(tag)">×</button>
      </span>
      <input #input
        (keydown.enter)="add(input.value); input.value=''"
        (keydown.comma)="add(input.value); input.value=''"
        (blur)="onTouched()" />
    </div>`
})
export class ChipSelectComponent
  implements ControlValueAccessor {

  selected: string[] = [];

  // Angular writes an array to us
  writeValue(v: string[] | null) {
    this.selected = v ?? [];
  }

  add(tag: string) {
    tag = tag.trim();
    if (tag && !this.selected.includes(tag)) {
      this.selected = [...this.selected, tag];
      this.onChange(this.selected);  // emit full array
    }
  }

  remove(tag: string) {
    this.selected = this.selected.filter(t => t !== tag);
    this.onChange(this.selected);
  }

  // Always emit immutable copies — never mutate
  // in place or Angular's change detection misses it

  private onChange   = (_: string[]) => {};
  private onTouched  = () => {};
  registerOnChange(fn: any)  { this.onChange = fn; }
  registerOnTouched(fn: any) { this.onTouched = fn; }
  setDisabledState(d: boolean) { this.isDisabled = d; }
}

7. Validation Integration — Validator & AsyncValidator

Sync Validator (Validator interface)

// Implement BOTH CVA + Validator in one component
@Component({
  providers: [
    { provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(()=>C), multi:true },
    // Also register as a validator:
    { provide: NG_VALIDATORS,      useExisting: forwardRef(()=>C), multi:true }
  ]
})
export class C implements ControlValueAccessor, Validator {

  // The validate() method is called by Angular whenever
  // the value changes and the form needs error state
  validate(ctrl: AbstractControl): ValidationErrors | null {
    const val = ctrl.value;
    if (!val) return { required: true };
    if (val < 1) return { min: { min: 1, actual: val } };
    if (val > 5) return { max: { max: 5, actual: val } };
    return null;  // valid
  }

  // Call this whenever your validation logic might change
  // (e.g., if validation depends on an @Input)
  private onValidatorChange = () => {};
  registerOnValidatorChange(fn: ()=>void) {
    this.onValidatorChange = fn;
  }
}

// In template — errors work automatically
<app-star-rating formControlName="rating"></app-star-rating>
<mat-error *ngIf="f.get('rating')?.hasError('required')">
  Rating is required
</mat-error>

Async Validator (AsyncValidator interface)

// Async: check username availability on server
@Component({
  providers: [
    { provide: NG_VALUE_ACCESSOR, useExisting: forwardRef(()=>C), multi:true },
    { provide: NG_ASYNC_VALIDATORS, useExisting: forwardRef(()=>C), multi:true }
  ]
})
export class C implements ControlValueAccessor, AsyncValidator {
  private http = inject(HttpClient);

  validate(ctrl: AbstractControl):
    Observable<ValidationErrors | null> {

    if (!ctrl.value) return of(null);

    return this.http.get<boolean>(
      `/api/check-username/${ctrl.value}`
    ).pipe(
      debounceTime(400),
      map(taken => taken ? { usernameTaken: true } : null),
      catchError(() => of(null))
    );
  }
}
// While async validator is pending:
// control.status === 'PENDING'
// Show a spinner in your template while status is PENDING

8. Accessing NgControl from Inside the CVA

Sometimes your CVA needs to read its own form control's state — e.g., to show an error border when the control is invalid and touched. You can inject NgControl, but you cannot also be registered in NG_VALUE_ACCESSOR from the same provider array — that creates a circular dependency. Use the injection approach below instead. 
// The correct pattern — inject NgControl instead of
// providing via NG_VALUE_ACCESSOR
@Component({
  selector: 'app-smart-input',
  standalone: true,
  // NO NG_VALUE_ACCESSOR here — we inject NgControl instead
})
export class SmartInputComponent
  implements ControlValueAccessor, OnInit {

  // inject with optional:true so it works in template-driven too
  private readonly ngControl = inject(NgControl, { optional: true });

  ngOnInit() {
    if (this.ngControl) {
      // Register ourselves as the value accessor
      this.ngControl.valueAccessor = this;
    }
  }

  // Now you can read control state in the template:
  get isInvalid(): boolean {
    return !!(
      this.ngControl?.invalid &&
      this.ngControl?.touched
    );
  }

  get errors() { return this.ngControl?.errors; }

  // template: [class.error]="isInvalid"
  private onChange   = (_: any) => {};
  private onTouched  = () => {};
  writeValue(v: any)        { ... }
  registerOnChange(fn: any)  { this.onChange = fn; }
  registerOnTouched(fn: any) { this.onTouched = fn; }
}

Why the circular dependency happens

If you both provide via NG_VALUE_ACCESSOR AND inject NgControl, Angular tries to create the component to satisfy NgControl, which needs NG_VALUE_ACCESSOR, which creates the component again — a cycle.

NgControl
needs NG_VALUE_ACCESSOR
⇆
YourComponent
injects NgControl

Circular! Error: Cannot instantiate cyclic dependency!

Resolution via ngOnInit

Fix: Remove the NG_VALUE_ACCESSOR provider. In ngOnInit(), assign ngControl.valueAccessor = this. Angular defers the accessor assignment until after construction, breaking the cycle.

9. Common Pitfalls

Forgetting to call onChange on every value change

If you update this.value internally but forget to call this.onChange(newVal), the form control never knows the value changed. The form stays stale.

// BAD — form never sees the change
handleClick(v: number) {
  this.value = v; // missing onChange!
}

// GOOD
handleClick(v: number) {
  this.value = v;
  this.onChange(v); // ← always required
}

Not implementing setDisabledState

When control.disable() is called, Angular calls setDisabledState(true). If you don't implement it, your UI stays interactive even though the form thinks it's disabled.

// Must implement — don't skip it
setDisabledState(isDisabled: boolean) {
  this.isDisabled = isDisabled;
  // Apply to native element too if needed:
  this.elementRef.nativeElement
    .setAttribute('aria-disabled', String(isDisabled));
}

Calling onChange inside writeValue

When Angular calls writeValue(), it's setting the value from the outside. Calling onChange() inside it creates an infinite update loop.

// BAD — infinite loop
writeValue(v: any) {
  this.value = v;
  this.onChange(v); // ← NEVER do this
}

// GOOD — just update internal state
writeValue(v: any) {
  this.value = v;
}

Mutating the array value in place

For array-valued CVAs (chip lists, multi-selects), mutating the array directly won't trigger Angular's change detection. Always create a new array reference.

// BAD — Angular doesn't see this change
add(item: string) {
  this.items.push(item);
  this.onChange(this.items);
}

// GOOD — new array reference
add(item: string) {
  this.items = [...this.items, item];
  this.onChange(this.items);
}

Missing forwardRef on the provider

The class is used inside its own decorator, before the class definition is complete. Without forwardRef, the value is undefined at the time the decorator runs.

// BAD — MyComponent is undefined here
providers: [{
  provide: NG_VALUE_ACCESSOR,
  useExisting: MyComponent, // undefined!
  multi: true
}]

// GOOD — lazily evaluated
providers: [{
  provide: NG_VALUE_ACCESSOR,
  useExisting: forwardRef(() => MyComponent),
  multi: true
}]

Not calling onTouched on blur

Without calling onTouched(), validators' error messages that rely on control.touched never appear. The standard is to show errors only after the user has interacted and left the field.

// In your template
(blur)="onTouched()"

// Or in component — trigger when focus leaves
handleBlur() {
  this.onTouched(); // marks control as touched
}

10. Testing a CVA Component

Testing a CVA is straightforward: wrap it in a minimal host component with a FormControl, use TestBed, and assert that the form control and the component's UI stay in sync in both directions. 
// star-rating.component.spec.ts
import { TestBed, ComponentFixture } from '@angular/core/testing';
import { ReactiveFormsModule, FormControl } from '@angular/forms';
import { StarRatingComponent } from './star-rating.component';

// Minimal host for testing in a form context
@Component({
  standalone: true,
  imports: [ReactiveFormsModule, StarRatingComponent],
  template: `<app-star-rating [formControl]="ctrl"></app-star-rating>`
})
class TestHost {
  ctrl = new FormControl<number | null>(null);
}

describe('StarRatingComponent CVA', () => {
  let host: TestHost;
  let fixture: ComponentFixture<TestHost>;

  beforeEach(async () => {
    await TestBed.configureTestingModule({
      imports: [TestHost]
    }).compileComponents();
    fixture = TestBed.createComponent(TestHost);
    host = fixture.componentInstance;
    fixture.detectChanges();
  });

  it('writeValue: form → component', () => {
    host.ctrl.setValue(4);
    fixture.detectChanges();
    // Assert the 4th star has the 'lit' class
    const stars = fixture.debugElement.queryAll(By.css('.star'));
    expect(stars[3].nativeElement.classList).toContain('lit');
    expect(stars[4].nativeElement.classList).not.toContain('lit');
  });

  it('onChange: component → form', () => {
    // Simulate clicking the 3rd star
    const stars = fixture.debugElement.queryAll(By.css('.star'));
    stars[2].triggerEventHandler('click', null);
    fixture.detectChanges();
    expect(host.ctrl.value).toBe(3);
    expect(host.ctrl.dirty).toBe(true);
  });

  it('setDisabledState: disables clicks', () => {
    host.ctrl.disable();
    fixture.detectChanges();
    const starComp = fixture.debugElement.query(
      By.directive(StarRatingComponent)
    ).componentInstance;
    expect(starComp.isDisabled).toBe(true);
  });

  it('validate: required error when null', () => {
    host.ctrl.setValue(null);
    expect(host.ctrl.hasError('required')).toBe(true);
    host.ctrl.setValue(3);
    expect(host.ctrl.valid).toBe(true);
  });
});
1 Create a test host component that wraps your CVA with a real FormControl.
2 Test writeValue by calling ctrl.setValue() and asserting the UI updates visually.
3 Test onChange by triggering a user interaction and asserting ctrl.value changed.
4 Test touched by triggering a blur event and asserting ctrl.touched === true.
5 Test disabled by calling ctrl.disable() and asserting clicks do nothing.
6 Test validators by setting boundary values and asserting ctrl.errors.
