RxJS Custom Operators

Build your own reusable, composable pipeable operators — from dead-simple to production-grade

Anatomy of a Pipeable Operator

The template — always the same shape

A pipeable operator is just a function that returns a function. The outer function takes config, the inner function takes an Observable and returns an Observable.

import { Observable, OperatorFunction } from 'rxjs';

// The universal template
function myOperator<T, R>(
  ...config: any[]
): OperatorFunction<T, R> {

  // Return a function: Observable<T> → Observable<R>
  return (source$: Observable<T>) => {
    return new Observable<R>(subscriber => {

      // Subscribe to the source
      const sub = source$.subscribe({
        next: value => {
          // Transform the value and emit to subscriber
          subscriber.next(value as unknown as R);
        },
        error: err => subscriber.error(err),
        complete: () => subscriber.complete()
      });

      // Teardown — called on unsubscribe
      return () => sub.unsubscribe();
    });
  };
}

Simpler: pipe existing operators together

Most custom operators don't need a raw new Observable(). You can compose existing operators with pipe() — RxJS does the heavy lifting.

import { pipe } from 'rxjs';
import { filter, map, debounceTime, distinctUntilChanged }
  from 'rxjs/operators';

// Custom operator: "search-ready" string stream
function searchReady(delayMs = 300) {
  return pipe(
    debounceTime(delayMs),
    distinctUntilChanged(),
    filter((v: string) => v.trim().length >= 2),
    map((v: string) => v.trim().toLowerCase())
  );
}

// Usage
searchInput$.pipe(
  searchReady(400),
  switchMap(q => http.get(`/search?q=${q}`))
).subscribe(...);

How pipe() + operators compose — step by step

1
Source Observable emits a raw value downstream. e.g. " HELLO " from an input
2
Your operator function receives the source Observable. It returns a NEW Observable that subscribes to the source internally.
3
In the next() handler you transform, filter, or delay the value. You decide when (and whether) to call subscriber.next(newValue).
4
Error and complete should always be forwarded unless your operator intentionally handles them (e.g. catchError, finalize).
5
Teardown function returned from the Observable constructor is called on unsubscribe — always clean up timers, inner subscriptions, listeners.

Your Custom Operator Library — 8 Real Operators

filterNil() — remove null/undefined

// Problem: Observable<User | null> forces null checks everywhere

function filterNil<T>(): OperatorFunction<T | null | undefined, T> {
  return filter(
    (v): v is T => v !== null && v !== undefined
  );
}

// Usage — now TypeScript knows it's always User
user$.pipe(
  filterNil()  // Observable<User | null> → Observable<User>
).subscribe(user => user.name); // ✓ no null check needed

debug(label) — console logging operator

function debug<T>(label: string): OperatorFunction<T, T> {
  return tap({
    next:     v => console.log(`[${label}] next:`, v),
    error:    e => console.error(`[${label}] error:`, e),
    complete: () => console.log(`[${label}] complete`),
  });
}

// Drop in anywhere in a pipe to inspect live values
search$.pipe(
  debug('raw input'),
  debounceTime(300),
  debug('after debounce'),
  switchMap(q => api.search(q)),
  debug('api result')
); // Remove in production — or use isProdEnvironment guard

tapOnce(fn) — side effect only on first value

function tapOnce<T>(fn: (v: T) => void): OperatorFunction<T, T> {
  return (source$) => new Observable<T>(sub => {
    let fired = false;
    return source$.subscribe({
      next: v => {
        if (!fired) { fired = true; fn(v); }
        sub.next(v);
      },
      error: e => sub.error(e),
      complete: () => sub.complete()
    });
  });
}

// Use case: start a spinner on the first result of a stream
data$.pipe(
  tapOnce(() => this.showWelcomeBanner()),
).subscribe(...);

retryWithDelay(n, ms) — exponential backoff

function retryWithDelay(
  times: number,
  delayMs: number,
  backoff = true
): MonoTypeOperatorFunction<any> {
  return (source$) =>
    source$.pipe(
      retryWhen(errors$ =>
        errors$.pipe(
          scan((attempt, err) => {
            if (attempt >= times) throw err;
            return attempt + 1;
          }, 0),
          switchMap(attempt => {
            const wait = backoff
              ? delayMs * Math.pow(2, attempt - 1)
              : delayMs;
            console.log(`Retry ${attempt}/${times} in ${wait}ms`);
            return timer(wait);
          })
        )
      )
    );
}

// Usage
http.get('/flaky-api').pipe(
  retryWithDelay(3, 1000) // retry 3×: 1s, 2s, 4s delays
).subscribe(...);

distinctUntilKeysChanged(keys)

function distinctUntilKeysChanged<T>(
  keys: (keyof T)[]
): MonoTypeOperatorFunction<T> {
  return distinctUntilChanged((a, b) =>
    keys.every(k => a[k] === b[k])
  );
}

// Only re-render if name or email changed (ignore updatedAt)
user$.pipe(
  distinctUntilKeysChanged(['name', 'email'])
).subscribe(user => this.renderProfile(user));
Demo — only emits when watched keys change

pluckKey(key) — type-safe property extractor

function pluckKey<T, K extends keyof T>(
  key: K
): OperatorFunction<T, T[K]> {
  return map(obj => obj[key]);
}

// Type-safe: TS knows result is string (User['name'])
user$.pipe(
  pluckKey('name')     // Observable<User> → Observable<string>
).subscribe(name => console.log(name.toUpperCase())); // ✓

// Compare: built-in pluck() loses types
user$.pipe(pluck('name')) // Observable<unknown> ← no type safety ❌

throttleMap(fn, delay) — rate-limited switchMap

// Like switchMap but enforces a minimum delay
// between emissions — prevents UI flickering
function throttleMap<T, R>(
  project: (v: T) => Observable<R>,
  minDelayMs: number
): OperatorFunction<T, R> {
  return (source$) =>
    source$.pipe(
      switchMap(v =>
        combineLatest([
          project(v),
          timer(minDelayMs)   // always wait at least minDelayMs
        ]).pipe(map(([result]) => result))
      )
    );
}

// Result never shows for less than 500ms
// (avoids loading flash on fast network)
search$.pipe(
  throttleMap(q => api.search(q), 500)
);

withLoading(signal) — auto loading state

function withLoading<T>(
  loadingSignal: WritableSignal<boolean>
): MonoTypeOperatorFunction<T> {
  return (source$) => new Observable<T>(sub => {
    loadingSignal.set(true);
    return source$.pipe(
      finalize(() => loadingSignal.set(false))
    ).subscribe(sub);
  });
}

// Automatically manages a loading signal
export class ProductsComponent {
  loading = signal(false);

  products$ = search$.pipe(
    switchMap(q => api.search(q).pipe(
      withLoading(this.loading) // set true → fetch → set false
    ))
  );
}

Testing Custom Operators

Unit test any custom operator with marble testing

import { TestScheduler } from 'rxjs/testing';

describe('filterNil()', () => {
  let scheduler: TestScheduler;

  beforeEach(() => {
    scheduler = new TestScheduler((actual, expected) =>
      expect(actual).toEqual(expected)
    );
  });

  it('should filter null and undefined values', () => {
    scheduler.run(({ cold, expectObservable }) => {
      const source$  = cold('a-b-c-d-e|', {
        a: 1, b: null, c: 2, d: undefined, e: 3
      });
      const expected = 'a---c---e|'; // nulls gapped out

      expectObservable(source$.pipe(filterNil()))
        .toBe(expected, { a: 1, c: 2, e: 3 });
    });
  });
});
