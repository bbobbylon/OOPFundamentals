Angular HTTP Interceptors — Advanced Patterns

Functional interceptors, HttpContextToken, auth refresh, caching, retry, and deduplication — with a step-through chain visualizer.

1. Class-Based vs Functional Interceptors

OLD: Class-Based (Angular 14-)
@Injectable()
export class AuthInterceptor
  implements HttpInterceptor {

  constructor(private auth: AuthService) {}

  intercept(
    req: HttpRequest<any>,
    next: HttpHandler
  ): Observable<HttpEvent<any>> {
    const token = this.auth.getToken();
    const cloned = req.clone({
      headers: req.headers.set(
        'Authorization', `Bearer ${token}`)
    });
    return next.handle(cloned);
  }
}

// Register:
{ provide: HTTP_INTERCEPTORS,
  useClass: AuthInterceptor, multi: true }
NEW: Functional (Angular 15+)
// Just a function — no class, no @Injectable
export const authInterceptor: HttpInterceptorFn =
  (req: HttpRequest<unknown>,
   next: HttpHandlerFn) => {

  const auth = inject(AuthService);
  const token = auth.getToken();

  if (!token) return next(req);

  return next(req.clone({
    headers: req.headers.set(
      'Authorization', `Bearer ${token}`)
  }));
};

// Register:
provideHttpClient(
  withInterceptors([authInterceptor])
)
Functional interceptors are the modern approach. They use inject() for dependencies, are tree-shakable, and don't require a class or @Injectable. The execution order matches the array order for requests and reverses for responses.

2. HttpContextToken — Per-Request Metadata

HttpContextToken lets you attach metadata to individual HTTP requests — without modifying headers or the URL. The interceptor reads this metadata to decide how to handle the request. 

import { HttpContextToken, HttpContext } from '@angular/common/http';

// Define tokens (factory = default value)
export const SKIP_AUTH = new HttpContextToken<boolean>(() => false);
export const CACHE_TTL = new HttpContextToken<number>(() => 0);
export const RETRY_COUNT = new HttpContextToken<number>(() => 3);

// Use in interceptor
export const authInterceptor: HttpInterceptorFn = (req, next) => {
  if (req.context.get(SKIP_AUTH)) {
    return next(req); // skip auth for this request
  }
  return next(req.clone({
    headers: req.headers.set('Authorization', 'Bearer ...')
  }));
};

// Set context at the call site
this.http.get('/api/public', {
  context: new HttpContext()
    .set(SKIP_AUTH, true)
    .set(CACHE_TTL, 60000)
});

3. Token Refresh Interceptor Pattern

When an API returns 401 Unauthorized, refresh the auth token and retry the original request. The tricky part: if multiple requests fail with 401 simultaneously, only perform one refresh — not one per request. 

let refreshing$ = null; // in-progress refresh or null

export const refreshInterceptor: HttpInterceptorFn = (req, next) => {
  const auth = inject(AuthService);

  return next(req).pipe(
    catchError(error => {
      if (error.status !== 401) return throwError(() => error);

      // Prevent multiple simultaneous refreshes
      if (!refreshing$) {
        refreshing$ = auth.refreshToken().pipe(
          tap(token => auth.setToken(token)),
          catchError(err => {
            refreshing$ = null;
            auth.logout();
            return throwError(() => err);
          }),
          finalize(() => refreshing$ = null),
          shareReplay(1)  // all waiting requests share this one refresh
        );
      }

      // Wait for refresh to complete, then retry with new token
      return refreshing$.pipe(
        switchMap(token => {
          const retried = req.clone({
            headers: req.headers.set('Authorization', `Bearer ${token}`)
          });
          return next(retried);
        })
      );
    })
  );
};
shareReplay(1) is the key: all requests waiting for the token refresh subscribe to the same Observable. The actual HTTP call to /auth/refresh happens exactly once, and all waiters get the result.

4. Cache Interceptor

interface CacheEntry {
  data: HttpResponse<unknown>;
  expiresAt: number;
}
const cache = new Map<string, CacheEntry>();

export const cacheInterceptor: HttpInterceptorFn = (req, next) => {
  // Only cache GET requests with CACHE_TTL set
  const ttl = req.context.get(CACHE_TTL);
  if (req.method !== 'GET' || !ttl) return next(req);

  const key = req.urlWithParams;
  const cached = cache.get(key);

  if (cached && cached.expiresAt > Date.now()) {
    return of(cached.data); // serve from cache
  }

  return next(req).pipe(
    tap(event => {
      if (event instanceof HttpResponse) {
        cache.set(key, {
          data: event,
          expiresAt: Date.now() + ttl
        });
      }
    })
  );
};

5. Retry with Exponential Backoff

import { retry, timer } from 'rxjs';

export const retryInterceptor: HttpInterceptorFn = (req, next) => {
  const maxRetries = req.context.get(RETRY_COUNT);

  return next(req).pipe(
    retry({
      count: maxRetries,
      delay: (error, attempt) => {
        // Don't retry client errors (4xx) — only network/server errors
        if (error.status >= 400 && error.status < 500) {
          return throwError(() => error);
        }
        // Exponential backoff: 1s, 2s, 4s
        const backoff = Math.pow(2, attempt - 1) * 1000;
        console.log(`Retry ${attempt} in ${backoff}ms`);
        return timer(backoff);
      }
    })
  );
};

6. Request Deduplication

const inFlight = new Map<string, Observable<HttpEvent<unknown>>>();

export const deduplicateInterceptor: HttpInterceptorFn = (req, next) => {
  if (req.method !== 'GET') return next(req);

  const key = req.urlWithParams;

  if (inFlight.has(key)) {
    // Return the same in-progress Observable
    return inFlight.get(key)!;
  }

  const request$ = next(req).pipe(
    shareReplay(1),
    finalize(() => inFlight.delete(key))
  );

  inFlight.set(key, request$);
  return request$;
};
Useful when multiple components make the same API call on init simultaneously. All get the same single response rather than flooding the server.
