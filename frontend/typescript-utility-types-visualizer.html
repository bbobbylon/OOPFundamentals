1. Why utility types matter

You almost never declare the "same" type twice in a well-typed codebase. A User exists once; UserUpdate is "every User field optional"; UserCreate is "every User field but the id". Utility types let you derive all those shapes from the single source of truth — so when User changes, every derived type follows.

type User = { id: string; name: string; email: string; admin: boolean };

type UserUpdate = Partial<User>;        // all fields optional
type UserCreate = Omit<User, 'id'>;       // drop the auto-assigned id
type UserPublic = Pick<User, 'id' | 'name'>; // safe-to-render fields
Mental model: utility types are functions at the type level. They take types in, give types out. The arguments go in <...> instead of (...) — that's the only real difference.

2. Object utilities

Partial<T> — every field optional

type Partial<T> = { [K in keyof T]?: T[K] };
type User      = { id: string; name: string; admin: boolean };
type UserPatch = Partial<User>;
// { id?: string; name?: string; admin?: boolean }

function update(id: string, patch: UserPatch) { /* PATCH /users/:id */ }
update('u_1', { name: 'Bob' }); // ✅ partial body OK

Use for: PATCH payloads, default-merging configs, optional override props.

Required<T> — every field mandatory

type Required<T> = { [K in keyof T]-?: T[K] };

The opposite of Partial. Note the -? — that's the "remove optionality" modifier in mapped types.

type Config         = { host?: string; port?: number; retries?: number };
type ResolvedConfig = Required<Config>;
// { host: string; port: number; retries: number }

function withDefaults(cfg: Config): ResolvedConfig {
  return { host: 'localhost', port: 80, retries: 3, ...cfg };
}

Readonly<T> — every field immutable

type Readonly<T> = { readonly [K in keyof T]: T[K] };
type Frozen = Readonly<{ x: number; y: number }>;
const p: Frozen = { x: 1, y: 2 };
p.x = 3;  // ❌ Cannot assign to 'x' because it is a read-only property

Caveat: shallow only. Readonly<{ inner: { a: number } }> still lets you mutate p.inner.a. Build your own DeepReadonly with a recursive mapped type — see §9.

Pick<T, K> — keep only these keys

type Pick<T, K extends keyof T> = { [P in K]: T[P] };
type User       = { id: string; name: string; email: string; password: string };
type UserPublic = Pick<User, 'id' | 'name' | 'email'>;
// { id: string; name: string; email: string }   ← password excluded

Use for: safely exposing a subset of a domain type to the outside world (API responses, view models).

Omit<T, K> — drop these keys

type Omit<T, K extends keyof any> = Pick<T, Exclude<keyof T, K>>;

The inverse of Pick. Note K extends keyof any — Omit lets you pass keys that aren't in T without error (controversial design choice).

type User       = { id: string; name: string; password: string };
type CreateUser = Omit<User, 'id'>;
// { name: string; password: string }   ← id is generated server-side

type Bad = Omit<User, 'idd'>; // ⚠ no error! typo silently does nothing

Record<K, V> — object with known keys, uniform values

type Record<K extends keyof any, V> = { [P in K]: V };
type Role        = 'admin' | 'user' | 'guest';
type Permissions = Record<Role, string[]>;
// { admin: string[]; user: string[]; guest: string[] }

const perms: Permissions = {
  admin: ['read', 'write', 'delete'],
  user:  ['read', 'write'],
  guest: ['read'],
};
// Forgetting a role → compile error. Adding a new role to Role → compile error here too. ✨

// Wide-key form for "any string maps to a value":
type Dict = Record<string, number>; // same as { [k: string]: number }

3. Union utilities

Exclude<T, U> — remove members of U from union T

type Exclude<T, U> = T extends U ? never : T;
type Status = 'pending' | 'active' | 'banned';
type Live   = Exclude<Status, 'banned'>;
// 'pending' | 'active'

type NonString = Exclude<string | number | boolean, string>;
// number | boolean

Extract<T, U> — keep only members of T assignable to U

type Extract<T, U> = T extends U ? T : never;
type Mixed  = string | number | { id: string };
type Strs   = Extract<Mixed, string>;        // string
type Object = Extract<Mixed, { id: string }>; // { id: string }

NonNullable<T> — remove null and undefined

type NonNullable<T> = T & {}; // modern; older: Exclude<T, null | undefined>
type Maybe = string | null | undefined;
type Defined = NonNullable<Maybe>; // string

4. Function utilities

ReturnType<F> — what a function gives back

type ReturnType<F> = F extends (...a: any) => infer R ? R : never;
function getUser() {
  return { id: 'u_1', name: 'Bob', lastLogin: new Date() };
}

type User = ReturnType<typeof getUser>;
// { id: string; name: string; lastLogin: Date }

// Now User stays in sync with getUser's body — no manual duplication.

The typeof grabs the function's type; ReturnType extracts the return slot. This pattern is everywhere in real codebases — it's why you can stop hand-writing types that mirror a function's output.

Parameters<F> — tuple of argument types

type Parameters<F> = F extends (...a: infer P) => any ? P : never;
function register(name: string, age: number, admin = false) {}
type RegArgs = Parameters<typeof register>;
// [name: string, age: number, admin?: boolean]

// Useful for wrapping/decorating:
function withLog<F extends (...a: any[]) => any>(fn: F) {
  return (...args: Parameters<F>): ReturnType<F> => {
    console.log('call', args);
    return fn(...args);
  };
}

ConstructorParameters<C> & InstanceType<C>

type ConstructorParameters<C> = C extends new (...a: infer P) => any ? P : never;
type InstanceType<C> = C extends new (...a: any) => infer R ? R : never;
class Service { constructor(public url: string, public timeout: number) {} }

type Args    = ConstructorParameters<typeof Service>; // [string, number]
type ServInst = InstanceType<typeof Service>;          // Service

function make<C extends new (...a: any[]) => any>(
  C: C, ...args: ConstructorParameters<C>
): InstanceType<C> {
  return new C(...args);
}

ThisParameterType<F> & OmitThisParameter<F>

Niche but handy when you're typing decorators or proxies. Extract or strip the this annotation.

function click(this: HTMLElement, e: MouseEvent) { this.focus(); }
type ThisT  = ThisParameterType<typeof click>;   // HTMLElement
type Plain  = OmitThisParameter<typeof click>;   // (e: MouseEvent) => void

5. Async utilities

Awaited<T> — unwrap Promise (recursively!)

// recursive — peels nested promises until a non-promise type
type A = Awaited<Promise<string>>;                  // string
type B = Awaited<Promise<Promise<number>>>;        // number  ← recursive
type C = Awaited<boolean | Promise<boolean>>;     // boolean

// Real use: derive the resolved type of an async function
async function fetchUser() { return { id: 'u_1', name: 'Bob' }; }
type User = Awaited<ReturnType<typeof fetchUser>>;
// { id: string; name: string }

6. String utilities

Uppercase, Lowercase, Capitalize, Uncapitalize

Apply to string-literal types. Real use: building precise template-literal types and event-handler maps.

type Greeting = 'hello world';

type U  = Uppercase<Greeting>;   // 'HELLO WORLD'
type L  = Lowercase<'HELLO'>;       // 'hello'
type C  = Capitalize<'click'>;      // 'Click'
type Uc = Uncapitalize<'Click'>;    // 'click'

// Real use — turn event names into handler-prop names:
type Events  = 'click' | 'hover' | 'submit';
type Handlers = { [K in Events as `on${Capitalize<K>}`]: () => void };
// { onClick: () => void; onHover: () => void; onSubmit: () => void }

7. Combining utilities

The real power emerges when you chain them. A few patterns you'll meet constantly:

Update payload — every field optional, except id required

type User       = { id: string; name: string; email: string; admin: boolean };
type UserUpdate = Pick<User, 'id'> & Partial<Omit<User, 'id'>>;
// { id: string } & Partial<{ name; email; admin }>
// → id required, rest optional

function patch(u: UserUpdate) {}
patch({ id: 'u_1', name: 'Bob' }); // ✅
patch({ name: 'Bob' });            // ❌ id missing

Pick + Record — a typed map keyed by some subset

type User     = { id: string; name: string; email: string };
type Indexed  = Record<User['id'], User>;
// { [k: string]: User } — but keyed by the id field's type

Async + Return — derive the resolved type of an HTTP call

const getUser = async (id: string) => {
  const r = await fetch(`/users/${id}`);
  return r.json() as Promise<{ id: string; name: string }>;
};

type User = Awaited<ReturnType<typeof getUser>>;
// { id: string; name: string } — chain of three utilities

9. Build your own

Once you internalize keyof, mapped types, and conditional types, you can write any utility you need. A starter kit:

DeepReadonly — recursive freeze

type DeepReadonly<T> = {
  readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K];
};

Mutable — strip readonly

type Mutable<T> = { -readonly [K in keyof T]: T[K] };

NullableFields — make every field T | null

type Nullable<T> = { [K in keyof T]: T[K] | null };

PickByValue — pick fields whose value is assignable to V

type PickByValue<T, V> = {
  [K in keyof T as T[K] extends V ? K : never]: T[K];
};

type User      = { id: number; name: string; admin: boolean; age: number };
type UserNums  = PickByValue<User, number>;
// { id: number; age: number }

OmitNever — drop keys whose value resolved to never

type OmitNever<T> = { [K in keyof T as T[K] extends never ? never : K]: T[K] };
