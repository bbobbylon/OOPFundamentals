app.config.ts

⚙️ What is app.config.ts — and why did Angular replace AppModule with a standalone ApplicationConfig?

app.config.ts is the entry point for configuring an Angular 17+ standalone application. It replaces the old AppModule pattern entirely. Instead of a class decorated with @NgModule({ imports: [...], providers: [...] }), you export a plain ApplicationConfig object with a providers array. This object is passed to bootstrapApplication(AppComponent, appConfig) in main.ts. Why the change? NgModule was Angular's original compilation and DI system, but it had problems: everything that needed to be available app-wide had to be imported through a module, modules could not be lazy-loaded themselves (only routes could), and the module system added complexity for sharing features. The standalone API removes NgModule from the equation — components, directives, and pipes are standalone by default and declare their own imports. app.config.ts sets up the root injector — the singleton DI scope that lives for the entire app lifetime. Key providers you'll use: provideRouter(routes) — sets up Angular Router with your routes. Additional feature functions: withComponentInputBinding() (maps route params to @Input), withViewTransitions() (CSS View Transitions API), withPreloading(PreloadAllModules). provideHttpClient() — sets up Angular's HttpClient. Crucial: add withInterceptors([...interceptors]) to register your auth interceptors (adds Bearer token to every request). provideAnimations() or provideAnimationsAsync() — Angular Material and animations require this. Injection Tokens: use { provide: MY_TOKEN, useValue: config } to make configuration (API URLs, feature flags) injectable via inject(MY_TOKEN).

app.config.ts Anatomy

The modern replacement for AppModule. Introduced in Angular 14 and made the default in Angular 17. Instead of decorating a class, you export an ApplicationConfig object and pass it to bootstrapApplication().

// src/app/app.config.ts
import { ApplicationConfig, provideZoneChangeDetection } from '@angular/core';
import { provideRouter, withComponentInputBinding, withViewTransitions,
         withPreloading, PreloadAllModules } from '@angular/router';
import { provideHttpClient, withInterceptors, withFetch } from '@angular/common/http';
import { provideAnimationsAsync } from '@angular/platform-browser/animations/async';
import { routes } from './app.routes';
import { authInterceptor } from './core/auth.interceptor';
import { loggingInterceptor } from './core/logging.interceptor';

export const appConfig: ApplicationConfig = {
  providers: [

    /* ── Change detection ──────────────────────────────────── */
    provideZoneChangeDetection({ eventCoalescing: true }),
    // eventCoalescing: batches multiple DOM events into one CD cycle

    /* ── Router ────────────────────────────────────────────── */
    provideRouter(
      routes,                          // your Route[] array
      withComponentInputBinding(),     // auto-bind :id param to @Input() id
      withViewTransitions(),           // browser View Transitions API (v17+)
      withPreloading(PreloadAllModules) // eager-preload lazy routes
    ),

    /* ── HTTP Client ────────────────────────────────────────── */
    provideHttpClient(
      withFetch(),                     // use fetch() instead of XMLHttpRequest
      withInterceptors([               // functional interceptors (array, ordered)
        authInterceptor,
        loggingInterceptor
      ])
    ),

    /* ── Animations ─────────────────────────────────────────── */
    provideAnimationsAsync(),          // lazy-load @angular/animations only when needed
    // provideAnimations()           ← synchronous alternative
    // provideNoopAnimations()       ← disable animations (useful in tests)

    /* ── App-level custom providers ─────────────────────────── */
    { provide: APP_BASE_HREF, useValue: '/' },
    { provide: API_URL, useValue: environment.apiUrl }
  ]
};

// src/main.ts
import { bootstrapApplication } from '@angular/platform-browser';
import { AppComponent } from './app/app.component';
import { appConfig } from './app/app.config';

bootstrapApplication(AppComponent, appConfig)
  .catch(err => console.error(err));

provideRouter — All Options

provideRouter(routes, ...features) configures the Angular router. Features are tree-shakable — unused ones don't add to bundle size.

Full provideRouter configuration

import {
  provideRouter,
  withComponentInputBinding,
  withViewTransitions,
  withPreloading,   PreloadAllModules,
  withHashLocation,
  withRouterConfig,
  withDebugTracing,
  withDisabledInitialNavigation,
  withEnabledBlockingInitialNavigation,
  withInMemoryScrolling,
  withNavigationErrorHandler
} from '@angular/router';

provideRouter(
  routes,

  /* Bind route params/queryParams to @Input() automatically */
  withComponentInputBinding(),

  /* Browser View Transitions API for route animations (v17+) */
  withViewTransitions({ skipInitialTransition: true }),

  /* Preload strategy */
  withPreloading(PreloadAllModules),
  // or: withPreloading(QuicklinkStrategy)  ← only visible links

  /* Use hash-based routing (#/path) — useful for static hosting */
  withHashLocation(),

  /* Scroll behavior */
  withInMemoryScrolling({
    scrollPositionRestoration: 'enabled', // restore scroll on back/forward
    anchorScrolling: 'enabled'            // handle #fragment links
  }),

  /* Global error handler for navigation failures */
  withNavigationErrorHandler(err => console.error(err)),

  /* Block initial navigation until app initializers complete */
  withEnabledBlockingInitialNavigation(),

  /* Log all router events (development only) */
  withDebugTracing()
)

Route definitions (app.routes.ts)

import { Routes } from '@angular/router';

export const routes: Routes = [
  {
    path: '',
    redirectTo: '/home',
    pathMatch: 'full'
  },
  {
    path: 'home',
    component: HomeComponent
  },
  {
    path: 'users/:id',
    component: UserDetailComponent,
    // withComponentInputBinding() auto-injects :id into @Input() id!
  },
  {
    path: 'admin',
    // Lazy load — the router downloads admin.routes.ts only when needed
    loadChildren: () => import('./admin/admin.routes').then(m => m.adminRoutes),
    canActivate: [authGuard],
    // Route-scoped providers — AdminService only alive in /admin subtree
    providers: [AdminService]
  },
  {
    path: '**',
    component: NotFoundComponent
  }
];

Functional guards (Angular 15+)

// auth.guard.ts — no class needed
import { inject } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';

export const authGuard: CanActivateFn = (route, state) => {
  const authService = inject(AuthService);
  const router = inject(Router);
  return authService.isLoggedIn()
    ? true
    : router.createUrlTree(['/login'], { queryParams: { returnUrl: state.url } });
};

provideHttpClient — Interceptors & Features

In the standalone world, HTTP interceptors are plain functions — no class, no @Injectable. They're passed directly to withInterceptors([]).

Functional interceptor pattern

// auth.interceptor.ts
import { inject } from '@angular/core';
import { HttpInterceptorFn } from '@angular/common/http';
import { AuthService } from './auth.service';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = inject(AuthService).getToken();
  if (!token) return next(req);

  const authed = req.clone({
    setHeaders: { Authorization: `Bearer ${token}` }
  });
  return next(authed);
};
// logging.interceptor.ts
export const loggingInterceptor: HttpInterceptorFn = (req, next) => {
  console.log(`→ ${req.method} ${req.url}`);
  return next(req).pipe(
    tap(event => {
      if (event instanceof HttpResponse)
        console.log(`← ${event.status} ${req.url}`);
    })
  );
};

provideHttpClient features

FeaturePurpose
withInterceptors([...])Functional interceptors in execution order
withFetch()Use the Fetch API instead of XHR — required for SSR compatibility
withRequestsMadeViaParent()Inherit interceptors from parent injector (for child routes)
withJsonpSupport()Enable JSONP requests via HttpClient.jsonp()
withXsrfConfiguration({...})Customize CSRF token header/cookie names
withNoXsrfProtection()Disable XSRF protection (APIs that use other CSRF strategies)

Interceptor ordering matters

// Interceptors run in array order for requests, reverse for responses:
withInterceptors([
  authInterceptor,      // 1st on request, 3rd on response
  cacheInterceptor,     // 2nd on request, 2nd on response
  loggingInterceptor    // 3rd on request, 1st on response (innermost)
])

// Request:  auth → cache → logging → server
// Response: server → logging → cache → auth

Class-based interceptors (legacy, still supported)

// If you have a class-based interceptor from an older NgModule:
import { withLegacyInterceptors } from '@angular/common/http';

provideHttpClient(
  withLegacyInterceptors(),  // enables HTTP_INTERCEPTORS multi-provider token
),
{ provide: HTTP_INTERCEPTORS, useClass: OldAuthInterceptor, multi: true }

Providers — All Forms

The providers array accepts several syntaxes, each for a different use case.

All provider syntaxes

import { ApplicationConfig } from '@angular/core';

export const appConfig: ApplicationConfig = {
  providers: [

    /* 1. Class shorthand — Angular creates a singleton */
    UserService,
    // same as: { provide: UserService, useClass: UserService }

    /* 2. useClass — provide a different implementation */
    { provide: AuthService, useClass: JwtAuthService },

    /* 3. useValue — provide a static value */
    { provide: LOCALE_ID, useValue: 'en-US' },
    { provide: APP_BASE_HREF, useValue: '/' },

    /* 4. useFactory — compute the value at injection time */
    {
      provide: LoggerService,
      useFactory: (env: Environment) => env.production
        ? new SilentLogger()
        : new ConsoleLogger(),
      deps: [Environment]
    },

    /* 5. useExisting — alias one token to another */
    { provide: AbstractUserService, useExisting: UserService },

    /* 6. multi: true — multiple providers for one token */
    { provide: APP_INITIALIZER, useFactory: initAuth, deps: [AuthService], multi: true },
    { provide: APP_INITIALIZER, useFactory: initI18n, deps: [I18nService], multi: true },

    /* 7. importProvidersFrom — import an NgModule's providers */
    importProvidersFrom(MatSnackBarModule, GoogleMapsModule),

    /* 8. provideExperimentalZonelessChangeDetection (v18+) */
    // provideExperimentalZonelessChangeDetection()
  ]
};

APP_INITIALIZER — run code before app starts

// Blocking initializers — app won't render until all complete
{
  provide: APP_INITIALIZER,
  useFactory: (auth: AuthService) => () => auth.loadUserFromStorage(),
  deps: [AuthService],
  multi: true          // multi: true required — there can be many initializers
}

// The factory returns a function () => Promise<any> | Observable<any> | void
// Angular awaits all APP_INITIALIZER promises before bootstrapping

importProvidersFrom — bridge to NgModules

import { importProvidersFrom } from '@angular/core';

// Use for any NgModule you can't yet convert to standalone providers:
importProvidersFrom(
  MatSnackBarModule,         // Angular Material dialogs
  HttpClientModule,          // if not using provideHttpClient directly
  TranslateModule.forRoot({  // ngx-translate (still NgModule-based)
    loader: { provide: TranslateLoader, useFactory: createTranslateLoader }
  })
)

Route-level providers — scoped services

// In app.routes.ts
{
  path: 'checkout',
  component: CheckoutComponent,
  providers: [
    CheckoutService,         // only alive while /checkout route is active
    PaymentService
  ]
}

Injection Tokens

Use InjectionToken to provide and inject values that aren't classes — configuration objects, primitives, factories.

Creating and providing tokens

// tokens.ts
import { InjectionToken } from '@angular/core';

// Basic token (no default)
export const API_URL = new InjectionToken<string>('API_URL');

// Token with a default factory (tree-shakable)
export const APP_CONFIG = new InjectionToken<AppConfig>('APP_CONFIG', {
  providedIn: 'root',
  factory: () => ({ theme: 'light', language: 'en' })
});

// Token for a primitive with environment config
export const MAX_RETRIES = new InjectionToken<number>('MAX_RETRIES');

// In app.config.ts providers:
{ provide: API_URL, useValue: environment.apiUrl },
{ provide: MAX_RETRIES, useValue: 3 }

Injecting tokens in services and components

// Class-based injection (constructor)
@Injectable({ providedIn: 'root' })
export class ApiService {
  constructor(@Inject(API_URL) private apiUrl: string) {}
}

// Function-based injection (inject() — Angular 14+)
@Injectable({ providedIn: 'root' })
export class ApiService {
  private apiUrl = inject(API_URL);

  getUsers() {
    return this.http.get(`${this.apiUrl}/users`);
  }
}

providedIn: 'root' vs app.config providers

providedIn: 'root'

@Injectable({ providedIn: 'root' })
export class UserService { }
  • Tree-shakable — removed from bundle if never injected
  • Singleton at root injector level
  • No need to list in providers array
  • Best for most services

app.config.ts providers

providers: [UserService]
// or:
{ provide: UserService, useClass: MockUserService }
  • Always included (not tree-shaken)
  • Useful for swapping implementations
  • Required for non-class values (primitives, tokens)
  • Required for APP_INITIALIZER, multi providers

Environment injection context — inject() outside constructor

// inject() can run in the injection context: constructor, field initializer,
// factory function, runInInjectionContext()

export function createHttpFactory(): HttpClient {
  return inject(HttpClient);  // ✓ inside a provider factory
}

export const myToken = new InjectionToken('myToken', {
  factory: () => inject(HttpClient)  // ✓ inside token factory
});

// Outside injection context — use runInInjectionContext:
const injector = inject(Injector);
injector.runInInjectionContext(() => {
  const svc = inject(SomeService);  // ✓
});

NgModule → Standalone Migration

The Angular CLI provides automatic migration schematics. Here's the before/after for the common patterns.

AppModule → app.config.ts

Before (NgModule)

@NgModule({
  imports: [
    BrowserModule,
    HttpClientModule,
    RouterModule.forRoot(routes),
    ReactiveFormsModule
  ],
  providers: [
    { provide: HTTP_INTERCEPTORS,
      useClass: AuthInterceptor, multi: true }
  ],
  bootstrap: [AppComponent]
})
export class AppModule { }

After (Standalone)

export const appConfig = {
  providers: [
    provideRouter(routes),
    provideHttpClient(
      withInterceptors([authInterceptor])
    ),
    provideReactiveFormsModule(),
  ]
};

// main.ts
bootstrapApplication(
  AppComponent, appConfig);

Running the migration schematic

# Full standalone migration — converts AppModule + all components
ng generate @angular/core:standalone

# Step-by-step approach (three passes):
# Pass 1: Convert components/pipes/directives to standalone
ng generate @angular/core:standalone --defaults

# Pass 2: Remove unnecessary NgModules
ng generate @angular/core:standalone --mode remove-modules

# Pass 3: Bootstrap using bootstrapApplication
ng generate @angular/core:standalone --mode bootstrap
